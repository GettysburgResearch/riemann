## 5. Memory-only bounded reproduction

The code has no inputs or file operations. It exhausts 3^3 and 3^2 labelled
allocations per chart and computes a_U by its defining finite subset sum.
All arithmetic is exact Python integers and Fraction; no rounding or analytic
oracle is used. Acceptance uses explicit exceptions and survives Python -O.
It verifies the two arithmetic charts, complete history counts, owner shares
and cancellation, not the missing native decoder.

```python
from fractions import Fraction
from itertools import product
from math import comb, gcd, isqrt, prod


def check(condition, message):
    if not condition:
        raise RuntimeError(message)


def prime(n):
    return n >= 2 and all(n % d for d in range(2, isqrt(n) + 1))


def a_u(support, cutoff):
    total = 0
    for bits in product((0, 1), repeat=len(support)):
        divisor = prod(p for p, bit in zip(support, bits) if bit)
        if divisor <= cutoff:
            total += (-1) ** sum(bits)
    return int(not support) - total


def histories(support, cutoff):
    result = []
    for assignment in product(range(3), repeat=len(support)):
        parts = [tuple(p for p, tag in zip(support, assignment) if tag == j)
                 for j in range(3)]
        coefficient = a_u(parts[0], cutoff) * a_u(parts[1], cutoff)
        coefficient *= (-1) ** len(parts[2])
        if coefficient:
            r, s, m = (prod(part) for part in parts)
            check(r > cutoff and s > cutoff, "illegal balanced history")
            check(r * s * m == prod(support), "product mismatch")
            result.append((r, s, m, coefficient))
    return result


charts = ((40, (41, 43, 47), (179, 223)),
          (64, (71, 73, 79), (401, 421)))
for U, left, right in charts:
    primes = (2, 3, 5, 7) + left + right
    check(len(primes) == len(set(primes)) and 67 not in primes, "label overlap")
    check(all(prime(p) for p in primes), "nonprime")
    Y, P, Q, g = U ** 6, 6, 35, 1
    c, d = prod(left), prod(right)
    N, M = P * c * c, Q * d * d
    check(gcd(c, d) == gcd(P * Q, c * d) == 1, "unclean cores")
    check(N <= 16 * Y and M <= 16 * Y, "physical horizon")
    check(Fraction(1, 8) < Fraction(N, M) < 8, "ratio-eight")
    x = Fraction(31 * Y, 16)
    check(Y < x < 2 * Y and 1 < N / x < 8 and 1 < M / x < 8,
          "no common interior kernel support")
    lhs, rhs = histories(left, U), histories(right, U)
    check(len(lhs) == 12 and len(rhs) == 2, "history coverage")
    check(sum(v for *_, v in lhs) == 0, "left cancellation")
    check(sum(v for *_, v in rhs) == 2, "right coefficient")
    ls, rs = Fraction(1, comb(5, 2)), Fraction(1, comb(4, 2))
    pairs = [lv * rv * ls * rs for *_, lv in lhs for *_, rv in rhs]
    check(len(pairs) == 24 and sum(pairs) == 0, "bilateral cancellation")
    check(pairs.count(Fraction(1, 60)) == 12
          and pairs.count(Fraction(-1, 60)) == 12, "exact shares")
    print("PASS", dict(U=U, dyadic=(U & (U - 1) == 0), Y=Y,
          P=P, Q=Q, g=g, c=c, d=d, N=N, M=M, limit=16 * Y,
          least_primes=(min(left), min(right)),
          histories=(len(lhs), len(rhs)), bilateral_histories=len(pairs),
          collapsed=sum(pairs)))
```
