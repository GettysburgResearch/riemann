\section{Exact Mellin conventions}

\begin{definition}[Reviewed tail Mellin transform]
\label{def:A-tail-mellin}
\lean{RiemannFormal.Analysis.tailMellin}
The conclusion-facing transform is defined directly as
\[
  \mathcal M_{\geq 1}f(s)
  =\int_{[1,\infty)} f(x)x^{-s-1}\,dx.
\]
It is separate from Mathlib's standard Mellin transform on
\((0,\infty)\).
\end{definition}

\begin{theorem}[Fixed logarithmic box]
\label{thm:A-log-box}
\lean{RiemannFormal.Analysis.logBoxMultiplier_nonzero}
For one fixed \(A>1\),
\[
  K_A(s)=\frac{1-A^{-s}}{s}
\]
is analytic and nonzero throughout \(\Re s>0\).
Semantic ID: \texttt{CONSUMER.MELLIN.ZERO\_SAFE\_BOX}.
The arithmetic positivity estimate for a smoothed detector remains a
separate producer obligation.
\end{theorem}

\begin{theorem}[Fixed holomorphic-defect transfer]

cannot remove a nonremovable singularity.
